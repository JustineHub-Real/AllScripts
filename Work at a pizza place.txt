-- Function to find FrontDoorMain → DoorTouch
local function findDoorTouch(houseName)
    local house = workspace.Houses:FindFirstChild(houseName)
    if house then
        for _, obj in ipairs(house:GetDescendants()) do
            if obj:IsA("Model") and obj.Name == "FrontDoorMain" then
                local doorTouch = obj:FindFirstChild("DoorTouch")
                if doorTouch and doorTouch:IsA("BasePart") then
                    return doorTouch
                end
            end
        end
    end
    return nil
end

local function countItemDuplicates(itemName)
    local count = 0
    for _, item in ipairs(backpack:GetChildren()) do
        if item:IsA("Tool") and item.Name == itemName then
            count += 1
        end
    end
    return count
end

local function equipItem(itemName)
    if char:FindFirstChild(itemName) then
        return true -- Already equipped
    end
    local item = backpack:FindFirstChild(itemName)
    if item then
        item.Parent = char
        print("Equipped:", itemName)
        return true
    end
    return false
end

local function unequipItem()
    for _, tool in ipairs(char:GetChildren()) do
        if tool:IsA("Tool") then
            tool.Parent = backpack
        end
    end
end

local function findDoorTouch(houseName)
    local house = workspace.Houses:FindFirstChild(houseName)
    if house then
        local frontDoor = house:FindFirstChild("FrontDoorMain", true)
        if frontDoor then
            return frontDoor:FindFirstChild("DoorTouch")
        end
    end
    return nil
end

function AutoDeliver()
    while AutoDeliverEnabled do
        for _, itemName in ipairs(houseItems) do
            if not AutoDeliverEnabled then return end -- Emergency exit

            local duplicateCount = countItemDuplicates(itemName)

            while duplicateCount > 0 and AutoDeliverEnabled do
                if equipItem(itemName) then
                    local houseName = houseLocations[itemName]
                    if houseName then
                        local doorTouch = findDoorTouch(houseName)
                        if doorTouch then
                            local lookAtCFrame = CFrame.lookAt(hrp.Position, doorTouch.Position)

                            -- Teleport and rotate smoothly
                            hrp:PivotTo(CFrame.new(doorTouch.Position + Vector3.new(0, 0, -2)))
                            task.wait(0.5) -- Small delay for smooth movement

                            print("Delivered to:", houseName)
                        end
                    end

                    -- Wait while checking if item disappears
                    local waitTime = 8
                    while waitTime > 0 and AutoDeliverEnabled do
                        task.wait(1)
                        waitTime -= 1

                        if not char:FindFirstChild(itemName) then
                            duplicateCount = countItemDuplicates(itemName)
                            if duplicateCount > 0 then
                                print("Item disappeared! Equipping next:", itemName)
                                equipItem(itemName)
                                waitTime = 8  -- Restart timer
                            else
                                break
                            end
                        end
                    end

                    -- Unequip item after delivery
                    unequipItem()
                    print("Item unequipped.")

                    -- Update duplicate count before looping again
                    duplicateCount = countItemDuplicates(itemName)
                else
                    break
                end
                task.wait(0.5) -- Prevents freezing by adding delay between actions
            end
        end
        task.wait(1) -- Prevents infinite fast looping
    end
end

local isTeleporting = false  -- Track toggle state

Roles:AddToggle({
    Name = "Auto Supplier Job",
    Description = "Automatically teleport to supply locations",
    Default = false,
    Callback = function(state)
        isTeleporting = state  -- Update toggle state
        if isTeleporting then
            task.spawn(function()  -- Run teleportation in a separate thread
                while isTeleporting do
                    local supplyButtons = workspace:FindFirstChild("SupplyButtons")
                    if supplyButtons then
                        -- Define teleport targets
                        local teleportTargets = {
                            supplyButtons:FindFirstChild("Button"),
                            supplyButtons:GetChildren()[4],
                            supplyButtons:GetChildren()[5],
                            supplyButtons:GetChildren()[7],
                            supplyButtons:GetChildren()[6],
                            supplyButtons:GetChildren()[3],
                            supplyButtons:FindFirstChild("MountainDew")
                        }

                        -- Get player's character
                        local player = game.Players.LocalPlayer
                        local character = player.Character
                        if not character or not character.PrimaryPart then return end

                        -- Teleport to each target safely
                        for _, target in ipairs(teleportTargets) do
                            if not isTeleporting then break end -- Stop if toggled off
                            if target and target:IsA("Model") then
                                local primaryPart = target.PrimaryPart or target:FindFirstChildOfClass("Part")
                                if primaryPart then
                                    -- Offset slightly above the target to prevent falling
                                    local newCFrame = primaryPart.CFrame + Vector3.new(0, 5, 0)
                                    character:SetPrimaryPartCFrame(newCFrame)

                                    -- Ensure the player doesn't clip through the ground
                                    task.wait(0.1)  -- Delay before teleporting to the next target
                                end
                            end
                        end
                    else
                        print("SupplyButtons folder not found!")
                    end
                end
            end)
        end
    end
})

-- Services
local Players = game:GetService("Players")
local LocalPlayer = Players.LocalPlayer
local SelectedPlayer = nil
local IsSpectating = false

-- Function to get sorted player names
local function GetSortedPlayerNames()
    local playerList = {}

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer then
            local displayName = player.DisplayName
            local userName = player.Name
            table.insert(playerList, displayName .. " (" .. userName .. ")")
        end
    end

    -- Sort the list alphabetically
    table.sort(playerList)
    return playerList
end

-- Add Section
Player:AddSection({"View Players"})

-- Refresh Button (Updates the dropdown manually)
Player:AddButton({
    Name = "Refresh the player",
    Description = "",
    Callback = function()
        local updatedPlayers = GetSortedPlayerNames()
        Player:UpdateDropdown("Choose Players", updatedPlayers) -- Update dropdown options
        Player:SetDropdownValue("Choose Players", updatedPlayers[1] or "") -- Set to first player or empty
        print("Player list refreshed")
    end
})

local noclipEnabled = false
local runService = game:GetService("RunService")

local function toggleNoclip(state)
    noclipEnabled = state

    runService.Stepped:Connect(function()
        if noclipEnabled and game.Players.LocalPlayer.Character then
            for _, part in pairs(game.Players.LocalPlayer.Character:GetDescendants()) do
                if part:IsA("BasePart") and part.CanCollide and not part:IsDescendantOf(game.Workspace.Terrain) then
                    part.CanCollide = false
                end
            end
        end
    end)
end

Player:AddToggle({
    Name = "Noclip",
    Default = false,
    Callback = function(state)
        toggleNoclip(state)
    end
})
